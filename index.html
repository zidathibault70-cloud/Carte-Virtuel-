<script>

let dbClients = [];

let tempUser = {};

let generatedOtp = "";

const DEV_SECRET = "DEV2026"; // Code secret du développeur


function hideAll() {

document.getElementById('step-home').classList.add('hidden');

document.getElementById('step-admin-login').classList.add('hidden');

document.getElementById('step-admin-dashboard').classList.add('hidden');

document.getElementById('step-client-info').classList.add('hidden');

document.getElementById('step-client-verify').classList.add('hidden');

document.getElementById('step-client-card').classList.add('hidden');

}


function goHome() {

hideAll();

document.getElementById('step-home').classList.remove('hidden');

}


function showClientLogin() {

hideAll();

document.getElementById('step-client-info').classList.remove('hidden');

}


function showAdminLogin() {

hideAll();

document.getElementById('step-admin-login').classList.remove('hidden');

}


function verifyAdmin() {

const code = document.getElementById('admin-code').value.trim();

if (code === DEV_SECRET) {

hideAll();

document.getElementById('step-admin-dashboard').classList.remove('hidden');

let statsHtml = `<b>Total Clients inscrits :</b> ${dbClients.length}<br><br>`;

if(dbClients.length === 0) {

statsHtml += "<i>Aucun client pour le moment.</i>";

} else {

dbClients.forEach((c, index) => {

statsHtml += `${index+1}. ${c.name} | Tél: ${c.phone} | CNI: ${c.cni} | Solde: $${c.balance}<br>`;

});

}

document.getElementById('admin-stats').innerHTML = statsHtml;

} else {

alert("Code développeur incorrect !");

}

}


function sendOtp() {

const name = document.getElementById('client-name').value.trim();

const phone = document.getElementById('client-phone').value.trim();

const email = document.getElementById('client-email').value.trim();

const channel = document.getElementById('otp-channel').value;


if(!name || !phone || !email) {

alert("Veuillez remplir tous les champs (Nom, Téléphone, Email).");

return;

}


tempUser = { name, phone, email };

generatedOtp = Math.floor(1000 + Math.random() * 9000).toString(); // Code à 4 chiffres


// Simulation de l'envoi du code

alert(`[SIMULATION ENVOI] Un code de confirmation a été envoyé par ${channel.toUpperCase()} !\n\nVotre code secret de test est : ${generatedOtp}`);


hideAll();

document.getElementById('step-client-verify').classList.remove('hidden');

document.getElementById('otp-sent-msg').innerText = `Code envoyé par ${channel} vers ${channel === 'sms' ? phone : email}.`;

}


function validateAndCreateAccount() {

const enteredOtp = document.getElementById('entered-otp').value.trim();

const cni = document.getElementById('client-cni').value.trim();


if(enteredOtp !== generatedOtp) {

alert("Code de confirmation erroné ! Veuillez réessayer.");

return;

}

if(!cni) {

alert("Le numéro de la pièce d'identité est obligatoire.");

return;

}


tempUser.cni = cni;

// Générer un faux numéro de carte USA (16 chiffres)

tempUser.cardNumber = "4" + Math.floor(100000000000000 + Math.random() * 900000000000000).toString().substring(0, 15);

tempUser.balance = "1,000.00"; // Solde de départ USD


dbClients.push(tempUser);


// Afficher la carte virtuelle

hideAll();

document.getElementById('step-client-card').classList.remove('hidden');

document.getElementById('card-num').innerText = tempUser.cardNumber.match(/.{1,4}/g).join(' ');

document.getElementById('card-holder').innerText = tempUser.name.toUpperCase();

document.getElementById('card-balance').innerText = tempUser.balance;

}

</script>
